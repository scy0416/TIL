## 오늘 배운 것
- DPO는 질문과 좋은 답, 나쁜 답을 전부 제공해서 학습한다.
- 보통 SFT를 했음에도 불구하고 모델의 응답이 아쉬울 때 사용한다.
- 손실은 두 답의 점수 차이가 기준보다 벌어졌는지를 확인한다.
- 기준 모델은 교안에서는 SFT를 완료한 모델이었으나, 시작점이 같으면 뭐로든 정할 수 있다.
- DPO에서 chosen은 선호하는 답이고, rejected는 비선호하는 답으로 chosen쪽이 답이 된다.
- 학습을 진행할 때에는 기본적으로 베이스 모델의 가중치를 얼려서 변하지 않게 하며, 애초에 ref값을 구할 때에만 사용된다.
- ref는 기준 모델의 chosen, rejected의 로그 확률 정보를 먼저 계산해서 가지고 있다.
- policy는 기준 모델에서 시작해서 학습된 모델을 의미하며, 특정한 파인튜닝이어야 하는 것은 아니다.
- policy에서도 chosen, rejected에 대한 확률 로그 정보를 계산해서 ref의 각 값을 빼면 chosen의 확률 로그에 대한 기준 대비 변화량과 rejected의 확률 로그에 대한 기준 대비 변화량을 계산한다.
- 이 때, chosen과 rejected의 확률 로그에 대한 기준 대비 변화량의 차가 margin이다.
- 이 값에 베타를 곱하고 목적 합수인 `-logsigmoid(margin)`을 통해서 loss를 계산한다.
- 이 loss값이 낮아져야 좋은 것이고, margin이 작아질수록 loss는 커지며, margin이 커질수록 loss는 작아진다.
- 이는 목적 함수가 `-logsigmoid`이기 때문이다.
- 만약 목적 함수에 음수가 없으면 margin은 작아지는 방향으로 학습이 된다.
- 점수는 확률 로그로써 답변 토큰들과 EOS까지의 확률 로그의 합이다.
- 베타는 learning rate랑은 다른 존재로, 베타의 값이 margin의 값을 억제하는 역할을 한다.
- 베타가 크면 margin이 커져 loss가 작아지고 베타가 작으면 margin이 작아져 loss가 커진다.
- wandb는 학습 셀 아래에 붙인다.
- 시각화를 해서 보면 loss는 낮아지고, margin은 커지기 때문에 좋아 보이지만 과최적화된 것을 확인할 수 있다.
- margin이 양수래도 그것이 성능의 개선이라고 확신할 수 없으며 오히려 과최적화에 의해서 답이 붕괴되었을 수도 있다.

## 왜 중요한가
- SFT로도 개선되지 않는 부분에 대해서 처리하는 방법론에 대한 것이기 때문에 중요하다.
- 특히 특정한 방향으로 학습되도록 유도하는 것이기 때문에 중요하다.

```python
# 응답 토큰 로그 확률 계산 및 margin 계산 예제 코드
# 응답 토큰의 로그확률을 합한 뒤 고정 기준 모델 대비 선호 차이를 계산합니다.
def response_logps(logits, labels):
    # logits: [B,T,V], labels: [B,T]. 위치 t가 t+1 토큰을 예측합니다.
    targets = labels[:, 1:]
    mask = targets.ne(-100)
    # print(targets)
    # print(mask)
    # print(labels)

    # 응답 위치만 골라 FP32 log-softmax와 같은 교차엔트로피 계산을 합니다.
    token_logps = -F.cross_entropy(
        logits[:, :-1, :][mask].float(),
        targets[mask],
        reduction="none",
    )

    # 선택한 토큰이 어느 응답에 속했는지 이용해 두 응답의 로그확률을 따로 합합니다.
    owners = mask.nonzero(as_tuple=False)[:, 0]
    sums = torch.zeros(
        logits.shape[0], device=logits.device
    ).scatter_add(0, owners, token_logps)
    return sums, mask.sum(dim=1)


def dpo_terms(policy_logps, reference_logps):
    # 한 쌍은 chosen, rejected 순서입니다. reference에는 기울기를 흘리지 않습니다.
    # 현재 정책의 점수에서 고정 기준의 점수를 빼 변화량을 비교합니다.
    log_ratios = policy_logps - reference_logps.detach()
    rewards = BETA * log_ratios
    margin = rewards[0] - rewards[1]

    # log-sigmoid를 직접 사용해 작은 확률에서도 계산을 안정적으로 유지합니다.
    return -F.logsigmoid(margin), margin, rewards


# 후보 6개에서 균등 분포일 때, 응답 토큰 2개와 1개의 로그확률 합은 다릅니다.
toy_logits = torch.zeros(2, 4, 6)
toy_labels = torch.tensor([[-100, -100, 2, 5], [-100, -100, 2, -100]])
toy_sums, toy_lengths = response_logps(toy_logits, toy_labels)
assert torch.allclose(
    toy_sums,
    -toy_lengths * math.log(6),
), '계산 결과가 비교 기준 또는 허용 오차와 다릅니다.'
print("응답 토큰 수:", toy_lengths.tolist())
print("로그확률 합:", toy_sums.tolist())
print("토큰 평균:", (toy_sums / toy_lengths).tolist())

reference_small = torch.tensor([-4.0, -3.0])
for policy_small in [
    torch.tensor([-4.0, -3.0]),
    torch.tensor([-3.0, -4.0]),
]:
    loss_small, margin_small, reward_small = dpo_terms(
        policy_small, reference_small
    )
    print(
        "정책:",
        policy_small.tolist(),
        "margin:",
        round(margin_small.item(), 6),
        "loss:",
        round(loss_small.item(), 6),
        "선호 확률:",
        round(torch.sigmoid(margin_small).item(), 6),
    )
assert torch.allclose(
    dpo_terms(reference_small, reference_small)[0],
    torch.tensor(math.log(2)),
), '계산 결과가 비교 기준 또는 허용 오차와 다릅니다.'
```

```python
# 갱신 전 모델의 점수를 미리 계산하고 학습 중에는 이 기준값을 바꾸지 않습니다.
def pair_logps(model, pair):
    batch = {
        name: values.to(model.device)
        for name, values in collate_pair(pair).items()
    }
    outputs = model(
        input_ids=batch["input_ids"],
        attention_mask=batch["attention_mask"],
        use_cache=False,
    )
    return response_logps(outputs.logits, batch["labels"])[0]
```

```python
# DPO 학습 예제 코드
# 기준 점수는 고정한 채 policy만 갱신합니다. 네 선호쌍마다 기울기를 모아 한 번 반영합니다.
# full-weight DPO입니다. 양자화나 LoRA 설정을 더하지 않아 목적함수의 변화에 집중합니다.
model.gradient_checkpointing_enable(
    gradient_checkpointing_kwargs={"use_reentrant": False}
)
model.train()
optimizer = torch.optim.AdamW(
    model.parameters(),
    lr=LEARNING_RATE,
    weight_decay=0.01,
    foreach=False,
)
sample_name = "model.layers.0.self_attn.q_proj.weight"
parameter = dict(model.named_parameters())[sample_name]
sample_indices = torch.linspace(
    0, parameter.numel() - 1, 256, device=parameter.device
).long()

# 별도 복사본을 남겨 학습 뒤에도 학습 전 가중치와 비교할 수 있게 합니다.
sample_before = (
    parameter.detach().flatten()[sample_indices].float().cpu().clone()
)
del parameter

# 다음 누적 묶음에 이전 기울기가 섞이지 않게 비웁니다.
optimizer.zero_grad(set_to_none=True)
torch.cuda.reset_peak_memory_stats()
training_started = time.perf_counter()
log_history, step_metrics, global_step = [], [], 0
for epoch in range(EPOCHS):
    order = list(range(len(pairs)))
    random.Random(SEED + epoch).shuffle(order)
    for position, index in enumerate(order, start=1):
        scores = pair_logps(model, pairs[index])
        loss, margin, rewards = dpo_terms(
            scores, reference_scores[index].to(model.device)
        )
        if not torch.isfinite(loss):
            raise FloatingPointError("DPO loss가 유한하지 않습니다.")

        # 한 쌍 안에서는 토큰 log-prob를 합하고, 네 쌍의 손실은 평균해 한 번 갱신합니다.
        (loss / ACCUMULATION).backward()
        step_metrics.append(
            (
                loss.item(),
                margin.item(),
                rewards[0].item(),
                rewards[1].item(),
            )
        )
        if position % ACCUMULATION == 0:
            # 가중치가 아니라 갱신 직전 기울기의 전체 크기를 제한합니다.
            grad_norm = torch.nn.utils.clip_grad_norm_(
                model.parameters(),
                max_norm=1.0,
                error_if_nonfinite=True,
            )

            # 네 쌍의 기울기를 반영한 뒤 비워 다음 묶음과 섞이지 않게 합니다.
            # 모아 둔 기울기로 가중치를 한 번 갱신합니다.
            optimizer.step()
            optimizer.zero_grad(set_to_none=True)
            global_step += 1
            means = [
                sum(row[column] for row in step_metrics)
                / len(step_metrics)
                for column in range(4)
            ]
            log_history.append(
                {
                    "step": global_step,
                    "epoch": epoch + position / len(order),
                    "loss": means[0],
                    "margin": means[1],
                    "chosen_reward": means[2],
                    "rejected_reward": means[3],
                    "grad_norm": grad_norm.item(),
                    "learning_rate": LEARNING_RATE,
                }
            )
            step_metrics.clear()
            if global_step % 4 == 0:
                print(
                    "갱신:",
                    global_step,
                    "구간 평균 loss:",
                    round(means[0], 6),
                    "margin:",
                    round(means[1], 6),
                )
assert (
    global_step == 24 and not step_metrics
), '24번 갱신을 마치지 못했거나 누적할 손실이 남았습니다.'
training_elapsed = time.perf_counter() - training_started

# 학습 중 기준은 그대로입니다.
assert torch.equal(
    reference_scores, reference_copy
), '학습 중 고정 reference 점수가 바뀌었습니다.'
parameter = dict(model.named_parameters())[sample_name]
sample_after = (
    parameter.detach().flatten()[sample_indices].float().cpu()
)
parameter_change = {
    "parameter": sample_name,
    "sample_count": 256,
    "changed_count": (sample_before != sample_after).sum().item(),
    "max_abs_change": (sample_after - sample_before).abs().max().item(),
}
assert (
    torch.isfinite(sample_after).all()
    and parameter_change["changed_count"] > 0
), '가중치 표본이 유한한 값인지, 실제로 갱신됐는지 확인하세요.'
del parameter, scores, loss, margin, rewards, optimizer
model.gradient_checkpointing_disable()
model.eval()
gc.collect()
torch.cuda.empty_cache()
print("가중치 표본 변화:", parameter_change)
print("학습 시간 초:", round(training_elapsed, 3))
```

## 헷갈렸던 부분 & 해결
- 특정한 무언가가 헷갈렸다기 보다는 그냥 전체적인 내용이 헷갈려서 클로드를 통해서 어느정도 틀을 잡고 튜터님께 이해한 내용을 확인받았다.
- 처음에는 margin을 계산하는 대상(chosen, rejected의 확률 로그의 기준 대비 변화량)을 단순히 chosen, rejected의 확률 정도로 생각했다.
- 그렇다기 보다는 baseline에서 학습된 policy가 baseline에 비해서 chosen, rejected를 답하도록 학습된 정도였다.
- 이를 통해서 더 높게 학습되던 낮게 학습되던 결국 서로의 거리만 멀게되게끔 학습이 된다는 것을 알게되었다.
- 그리고 최종 loss계산에서 기존의 baseline기준으로 망가뜨리고 싶지 않은 질문셋을 만들어서 그 질문에 대한 확률 로그를 점수로 하는 loss를 함께 계산해서 표현력을 어느정도 유지하는 것이 가능하지 않을까 하는 방법론에 대한 질문을 튜터님께 했다.
- 튜터님은 바로 하기보다는 정성스런 삽질이 될 수 있으니까 우선적으로 다른 사람들이 해본 방법들을 조사해보고서 시도해보라고 하셨다.
- 틀린말은 아니다.
- 내가 했던 고민을 아무도 안했을리는 없다.
