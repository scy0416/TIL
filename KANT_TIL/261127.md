def file_sha256(path):
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def json_sha256(value):
    # 키의 순서나 들여쓰기 대신 실제 설정 값을 비교합니다.
    canonical = json.dumps(
        value, ensure_ascii=False, sort_keys=True, separators=(",", ":")
    )
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


def write_json(path, value):
    Path(path).write_text(
        json.dumps(value, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )


def extract_zip(archive_path, destination, required_prefix=None):
    destination = Path(destination).resolve()
    if destination.exists():
        raise FileExistsError(
            f"이전 결과를 보관하고 빈 작업 폴더에서 시작하세요: {destination.name}"
        )
    with zipfile.ZipFile(archive_path) as archive:
        names = archive.namelist()
        if len(names) != len(set(names)):
            raise ValueError("중복된 ZIP 경로를 허용하지 않습니다.")
        for member in archive.infolist():
            target = (destination / member.filename).resolve()

            # 압축 파일의 이름이 복원 폴더 바깥으로 나가는 경로인지 먼저 검사합니다.
            if not target.is_relative_to(destination):
                raise ValueError(
                    "ZIP 경로가 복원 폴더 밖을 가리킵니다."
                )
            if (
                required_prefix is not None
                and not member.filename.startswith(required_prefix)
            ):
                raise ValueError(
                    f"ZIP에 필요한 상위 폴더는 {required_prefix}입니다."
                )
        archive.extractall(destination)
    return destination
