```python
# seed 고정 함수 예제
import random
import numpy as np
import torch


def set_seed(seed=42):
    """
    Python, NumPy, PyTorch의 random seed를 고정합니다.

    seed:
        실험을 재현하기 위해 사용할 정수값입니다.
    """

    # Python 기본 random 모듈의 seed를 고정합니다.
    random.seed(seed)

    # NumPy random seed를 고정합니다.
    np.random.seed(seed)

    # PyTorch CPU random seed를 고정합니다.
    torch.manual_seed(seed)

    # CUDA GPU를 사용할 수 있다면 CUDA random seed도 고정합니다.
    if torch.cuda.is_available():
        torch.cuda.manual_seed(seed)
        torch.cuda.manual_seed_all(seed)
```

```python
# config.json 저장 예제
import json
import platform
from pathlib import Path

import torch


def save_config(config, save_dir):
    """
    실험 설정 dictionary를 config.json 파일로 저장합니다.

    config:
        실험 설정을 담은 dictionary입니다.

    save_dir:
        config.json을 저장할 폴더입니다.
    """

    # 문자열 경로나 Path 객체를 모두 Path 객체로 변환합니다.
    save_dir = Path(save_dir)

    # 저장 폴더가 없으면 생성합니다.
    save_dir.mkdir(parents=True, exist_ok=True)

    # 저장할 파일 경로입니다.
    config_path = save_dir / "config.json"

    # ensure_ascii=False는 한글이 깨지지 않게 저장하기 위한 옵션입니다.
    # indent=2는 사람이 읽기 좋게 들여쓰기합니다.
    with open(config_path, "w", encoding="utf-8") as f:
        json.dump(config, f, ensure_ascii=False, indent=2)

    print(f"config saved to:{config_path}")
```

```python
import csv
import math
from pathlib import Path


METRIC_COLUMNS = [
    "epoch",
    "train_loss",
    "train_acc",
    "valid_loss",
    "valid_acc",
]


def init_metric_csv(save_dir, resume=False):
    """새 로그를 만들거나, 재개할 기존 로그를 확인합니다."""

    save_dir = Path(save_dir)
    save_dir.mkdir(parents=True, exist_ok=True)
    csv_path = save_dir / "metrics.csv"

    if resume:
        if not csv_path.exists():
            raise FileNotFoundError(f"metric log not found: {csv_path}")
        return csv_path

    if csv_path.exists():
        raise FileExistsError(
            f"기존 로그를 덮어쓰지 않습니다: {csv_path}"
        )

    with open(csv_path, "w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file)
        writer.writerow(METRIC_COLUMNS)

    return csv_path
```

```python
# 간단한 MLP의 state_dict 확인 예제
import torch
import torch.nn as nn

class SimpleMLP(nn.Module):
    def __init__(self, input_dim=4, hidden_dim=32, num_classes=3):
        super().__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_dim, num_classes)

    def forward(self, x):
        x = self.fc1(x)
        x = self.relu(x)
        logits = self.fc2(x)
        return logits

model = SimpleMLP()

# model.state_dict()는 parameter 이름과 Tensor를 담은 dictionary입니다.
model_state = model.state_dict()

for name, tensor in model_state.items():
    print(name, tensor.shape)
```

```python
# loss curve 시각화 예제
import matplotlib.pyplot as plt

def plot_loss_curves(history):
    """
    train loss와 validation loss를 한 그래프에 그리는 함수입니다.

    history:
        epoch별 metric이 저장된 딕셔너리입니다.
        반드시 "train_loss", "valid_loss" 키를 가지고 있어야 합니다.
    """

    # epoch 번호를 만듭니다.
    # history["train_loss"] 길이가 5라면 epoch는 [1, 2, 3, 4, 5]가 됩니다.
    epochs = range(1, len(history["train_loss"]) + 1)

    # 그래프 크기를 지정합니다.
    # 노션이나 Colab에서 너무 작게 보이지 않도록 적당히 키웁니다.
    plt.figure(figsize=(8, 5))

    # train loss 곡선을 그립니다.
    plt.plot(epochs, history["train_loss"], marker="o", label="train_loss")

    # validation loss 곡선을 그립니다.
    plt.plot(epochs, history["valid_loss"], marker="o", label="valid_loss")

    # 그래프 제목과 축 이름을 붙입니다.
    plt.title("Loss Curves")
    plt.xlabel("Epoch")
    plt.ylabel("Loss")

    # 범례와 격자를 표시합니다.
    plt.legend()
    plt.grid(True)

    # 그래프를 화면에 출력합니다.
    plt.show()
```

## 헷갈렸던 부분 & 해결
- 데이터셋과 데이터로더에도 state_dict가 있을 것이라고 생각했지만 실제로는 없었는데, 이는 state_dict는 학습 상태에 대한 정보(바뀌는 정보)이고, 데이터셋과 데이터로더는 데이터에 대한 접근 방법에 대한 정보이기 때문이었다.
- 데이터로더의 사용 배치에 대한 내용과 질문
- 교안에는 정보들을 저장하고 로드해서 복구하는 과정에서 데이터로더의 배치를 세팅하는 과정이 없어서 그래도 알아서 처리되는 것이라고 생각했다.
- 하지만 알아보니 제네레이터는 추출하는 설정에 관한 것으로 추출하는 배치 순서는 조절할 수 없는 것이었고, 이에 이미 소모한 배치를 비워야 하지 않냐는 의문을 제기했다.
- 이는 생각해보니 확신을 가지게 되었다.
- 사용한 배치를 다시 사용한다면 그만큼 사용되었어야 하지만 사용되지 않는 배치가 발생하게되고, 이는 사용되지 못한 배치들에 대한 표현력 학습의 기회를 박탈당하는 것이며, 이미 사용한 배치를 재사용하는 것에 대한 과적합 위험이 발생하게 된다.
- 또한 애초에 끊김없이 진행되었을 때와 로드했을 때의 사용하는 배치 종류가 달라지기 때문에 복원 자체가 제대로 이뤄지지 않았다고 보는 것이 옳다.
- 내 생각은 맞았지만 이건 에포크 중간에 저장 및 로드하는 경우이고, 에포크 경계에 대해서 저장 및 로드하는 경에는 교안이 맞다.
- 나의 의문은 에포크 중간에 배치가 소모되지 않았을 때에 연결해서 소모해야하는 것이었기 때문에 여기에 대한 질문으로는 적합하지 않았던 것이다.
