```python
# 쿼리 포스트 요청 예제
from fastapi import FastAPI, HTTPException


app = FastAPI(title="Beginner Query API")


@app.post("/query", response_model=QueryResponse)
def query(request: QueryRequest) -> QueryResponse:
    """한 개의 프롬프트를 선택된 호출 방식에 전달합니다."""

    # 7-1강에서 만든 규칙으로 프롬프트와 실제 출처를 함께 준비합니다.
    prompt, sources = make_prompt(request)
    # 설정이 없으면 누구나 실행 가능한 mock을 사용합니다.
    mode = os.getenv("LLM_MODE", "mock").strip().lower()

    if mode == "mock":
        answer = deterministic_mock(prompt)
        return QueryResponse(answer=answer, sources=sources, provider="mock")

    if mode == "ollama":
        api_url = os.getenv("OLLAMA_API_URL", "http://localhost:11434/api/chat").strip()
        model_name = os.getenv("OLLAMA_MODEL", "").strip()
        if not api_url or not model_name:
            raise HTTPException(status_code=503, detail="Ollama URL과 모델명을 확인하세요.")
        try:
            with httpx.Client(timeout=30.0) as client:
                answer = call_ollama(client, prompt, api_url, model_name)
        except (httpx.HTTPError, ValueError) as error:
            raise HTTPException(status_code=502, detail="모델 답변을 가져오지 못했습니다.") from error
        return QueryResponse(answer=answer, sources=sources, provider="ollama")

    if mode != "external":
        raise HTTPException(status_code=503, detail="LLM_MODE를 확인하세요.")
    # external 분기의 URL·키 검사와 호출 코드는 앞 절의 함수를 사용합니다.
    return call_external_mode(prompt)
```

## 헷갈렸던 부분 & 해결
- pydantic의 model_validator에 대해서 보다가 분명 클래스 메소드로 정의된걸 본 기억이 있는데, 인스턴스 메소드로 정의된 코드도 있어서 왜그런지 헷갈렸다.
- 클로드에 물어본 결과, mode가 after라면 이미 필드마다의 검증이 끝나고, 필드 사이의 관계에 대해서 처리할 때 주로 사용하고, before는 데이터 자체를 처리할 때 사용한다고 했다.
- before는 인스턴스가 생기기 전의 데이터에 대해서 처리하는 것이기 때문에 클래스 메소드여야 하는 것이다.
- 이 두 모드는 한 클래스 안에 동시에 존재할 수 있으며, 한 모드의 검증기를 여러 개 선언할 수도 있다고 한다.
- 파이썬 비동기에 대해서 조금 헷갈리고 있었다.
- 그래서 클로드를 통해서 이해를 해보는 과정을 가졌는데, 결과적으로 파이썬 파일을 실행하면 비동기 함수의 실행을 위해서 `asyncio.run()`를 통해서 이벤트 루프를 적어도 한 개를 만들어줘야 한다.
- 비동기 함수를 실행하면 반환되는 값은 결과가 아니라 코루틴으로 그 상태로는 실행되지 않는다.
- 그렇기 때문에 await나 gather를 사용하는 것이다.
- 이 때, await는 yield를 사용하는 것처럼 코루틴을 직접 실행시키고 결과를 받아내며, gather는 코루틴을 받으면 `ensure_future`를 통해서 태스크로 감싸게 된다.
- 그리고 gather는 future객체를 반환하고, await는 이 future를 await하며 코루틴들의 결과를 기다리게 된다.
- await는 코루틴 하나에 대해서만 처리하기 때문에 비동기라는 특성이 그렇게 크지 않다.
- 그렇기 때문에 gather를 통해서 코루틴들을 한 번에 처리하도록 만들어야 비동기적인 동작을 구현할 수 있는  것이다.
