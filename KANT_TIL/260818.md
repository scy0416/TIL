## 코드/예시
```python
# 5-1 예제
# [1] 이번 실습 전체에서 사용할 라이브러리를 불러옵니다.
# 아래 다섯 코드 블록은 같은 런타임에서 위에서 아래로 이어서 실행합니다.
import pandas as pd
from sklearn.datasets import make_classification
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import average_precision_score
from sklearn.model_selection import (
    GridSearchCV,
    RandomizedSearchCV,
    StratifiedKFold,
    train_test_split,
)

# [2] 실제 LLM 로그 대신 재현 가능한 모의 분류 데이터를 만듭니다.
# X는 14개의 수치 특성이고, y는 정답 label입니다.
# 이 실습에서 y=1은 needs_review, 즉 사람이 검토해야 하는 답변입니다.
X, y = make_classification(
    n_samples=1600,        # 전체 답변 수
    n_features=14,         # 모델이 입력으로 사용할 수치 특성 수
    n_informative=8,       # 정답 구분에 실제 정보를 주는 특성 수
    n_redundant=2,         # informative 특성에서 파생된 중복 특성 수
    weights=[0.88, 0.12],  # needs_review=0 약 88%, needs_review=1 약 12%
    class_sep=0.9,         # 두 클래스가 구분되는 정도
    random_state=42,       # 실행할 때마다 같은 데이터를 만들기 위한 seed
)

# [3] 후보 탐색용 개발 데이터와 최종 평가용 test를 먼저 분리합니다.
# stratify=y는 두 데이터에서 needs_review=1 비율이 크게 달라지지 않게 합니다.
# X_test와 y_test는 만들어 두지만, 후보 선택이 끝날 때까지 성능 계산에 쓰지 않습니다.
X_dev, X_test, y_dev, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    stratify=y,
    random_state=42,
)

# [4] 모든 후보가 공유할 4-fold CV를 만듭니다.
# shuffle=True로 섞고 seed=42를 사용해 같은 네 fold를 재현합니다.
# Grid와 Random이 같은 cv 객체를 사용해야 분할 차이가 점수 차이에 섞이지 않습니다.
cv = StratifiedKFold(
    n_splits=4,
    shuffle=True,
    random_state=42,
)

# [5] 탐색 결과를 보기 전에 최종 후보 선정 규칙도 코드로 고정합니다.
# 최고 평균 AP에서 0.01 이내인 후보만 남긴 뒤,
# fold 변동 → 트리 수 → 깊이 → 탐색법 이름 순서로 동률을 풉니다.
# 0.01은 AP의 절대 차이이며 통계적 유의성 기준은 아닙니다.
MEAN_TOLERANCE = 0.01
SELECTION_ORDER = [
    "std_AP", "n_estimators", "max_depth", "search"
]

# [6] 모든 후보가 공유할 기본 Random Forest를 정의합니다.
# class_weight="balanced"는 드문 needs_review=1에 더 큰 학습 가중치를 줍니다.
# n_jobs=1은 한 모델 안의 트리 학습을 순차 실행하도록 고정합니다.
model = RandomForestClassifier(
    class_weight="balanced",
    random_state=42,
    n_jobs=1,
)

# [7] 데이터 역할이 의도대로 나뉘었는지 최소 정보만 확인합니다.
# test에서는 행 수만 확인하며 예측이나 성능 지표는 계산하지 않습니다.
print("dev/test rows:", len(y_dev), len(y_test))
print("needs_review rate:", round(y_dev.mean(), 3))

# [1] Grid Search를 정의합니다.
# n_estimators 3개 × max_depth 4개이므로 총 12개 조합을 빠짐없이 평가합니다.
grid = GridSearchCV(
    model,
    {
        "n_estimators": [80, 140, 200],  # Random Forest를 구성할 트리 수
        "max_depth": [4, 6, 8, 10],       # 각 트리의 최대 깊이
    },
    scoring="average_precision",  # 모든 후보를 AP로 비교합니다.
    cv=cv,                        # 앞에서 만든 동일한 4-fold를 사용합니다.
    n_jobs=1,                     # 후보별 CV 학습을 순차 실행합니다.
    refit=False,                  # 탐색 직후 자동 재학습하지 않습니다.
)

# [2] Random Search를 정의합니다.
# 전체 이산 목록은 13 × 7 = 91개 조합이며 그중 n_iter=12개만 뽑습니다.
# Grid와 최소·최대 경계, 후보 수, CV, 지표를 맞추되 실제 좌표는 다릅니다.
random = RandomizedSearchCV(
    model,
    {
        "n_estimators": list(range(80, 201, 10)),  # 80~200, 10 간격
        "max_depth": list(range(4, 11)),            # 4~10, 1 간격
    },
    n_iter=12,                    # Grid와 동일하게 12개 후보만 평가합니다.
    scoring="average_precision", # Grid와 동일한 AP를 사용합니다.
    cv=cv,                       # Grid와 동일한 4-fold를 사용합니다.
    n_jobs=1,
    random_state=42,             # 매번 같은 Random 후보를 뽑기 위한 seed
    refit=False,                 # 이 단계에서는 후보별 CV 결과만 저장합니다.
)

# [3] 두 탐색 객체를 한 구조에 묶어 같은 방식으로 실행합니다.
searches = {
    "Grid": grid,
    "Random": random,
}

# [4] 두 탐색 모두 개발 데이터만 사용합니다.
# 후보 하나는 네 fold에서 네 번 학습됩니다.
# 따라서 탐색법당 12 × 4 = 48 fits, 두 탐색 합계는 96 CV fits입니다.
for search in searches.values():
    search.fit(X_dev, y_dev)

# [1] 탐색법별 최고 후보를 비교할 요약 행을 모읍니다.
rows = []

for name, search in searches.items():
    # cv_results_에는 후보별 파라미터, fold 점수, 학습 시간 등이
    # 같은 행 순서로 저장되어 있습니다.
    # best_index_는 평균 AP가 가장 높은 후보 행의 위치입니다.
    best_index = search.best_index_

    # params 배열의 길이는 실제로 평가한 후보 수입니다.
    candidate_count = len(search.cv_results_["params"])

    rows.append({
        "search": name,
        "candidates": candidate_count,

        # 후보마다 네 fold에서 학습되므로 CV fit 수는 후보 수 × fold 수입니다.
        "CV_fits": candidate_count * cv.get_n_splits(),

        # 최고 후보의 네 validation fold 평균 AP입니다.
        "best_mean_AP": search.cv_results_["mean_test_score"][best_index],

        # 최고 후보의 fold 점수가 분할에 따라 얼마나 흔들렸는지 나타냅니다.
        # 일반화 오차의 신뢰구간이 아니라 fold 간 변동의 요약입니다.
        "best_std_AP": search.cv_results_["std_test_score"][best_index],

        # 최고 후보가 사용한 n_estimators와 max_depth입니다.
        "best_params": search.cv_results_["params"][best_index],
    })

# [2] 탐색법별 최고 후보를 한 표로 만듭니다.
comparison = pd.DataFrame(rows)

# 평균 AP와 fold 표준편차는 소수점 셋째 자리까지 표시합니다.
print(comparison.to_string(
    index=False,
    formatters={
        "best_mean_AP": "{:.3f}".format,
        "best_std_AP": "{:.3f}".format,
    },
))
print("total CV fits:", comparison["CV_fits"].sum())

# [3] 점수와 별개로 후보 평가 횟수 계약이 지켜졌는지 검사합니다.
# Grid 12개, Random 12개, 탐색법당 48 fits, 전체 96 fits여야 합니다.
assert comparison["candidates"].tolist() == [12, 12]
assert comparison["CV_fits"].tolist() == [48, 48]
assert comparison["CV_fits"].sum() == 96

# [1] 이 함수는 학습이 끝난 탐색 객체만 입력받습니다.
# X_test와 y_test는 인자로 받지 않으며 함수 안에서도 참조하지 않습니다.
def select_candidate(
    fitted_searches,
    mean_tolerance=MEAN_TOLERANCE,
):
    candidate_rows = []

    # [2] 각 탐색의 최고 후보만이 아니라,
    # Grid와 Random이 평가한 전체 24개 후보 행을 한 표로 합칩니다.
    for search_name, fitted_search in fitted_searches.items():
        results = pd.DataFrame(fitted_search.cv_results_)

        for _, row in results.iterrows():
            candidate_rows.append({
                # 어떤 탐색법이 이 좌표를 평가했는지 기록합니다.
                "search": search_name,

                # 해당 후보의 4-fold 평균 AP와 fold 표준편차입니다.
                "mean_AP": row["mean_test_score"],
                "std_AP": row["std_test_score"],

                # 비교와 출력을 위해 param_* 값을 정수로 바꿉니다.
                "n_estimators": int(row["param_n_estimators"]),
                "max_depth": int(row["param_max_depth"]),
            })

    candidates = pd.DataFrame(candidate_rows)

    # [3] 전체 후보 중 가장 높은 평균 AP를 찾습니다.
    best_mean = candidates["mean_AP"].max()

    # 결과 전에 고정한 허용폭 안의 후보만 남깁니다.
    # 0.01은 최고 점수의 1%가 아니라 AP의 절대 차이 0.01입니다.
    eligible = candidates[
        candidates["mean_AP"] >= best_mean - mean_tolerance
    ]

    # [4] 역시 결과 전에 고정한 순서로 최종 후보 하나를 고릅니다.
    # fold 변동 → 트리 수 → 깊이 → 탐색법 이름 순서입니다.
    # mergesort는 모든 정렬 값이 같을 때 기존 순서도 안정적으로 유지합니다.
    selected = eligible.sort_values(
        SELECTION_ORDER,
        kind="mergesort",
    ).iloc[0]

    # 선택 후보, 허용 범위 후보 수, 전체 최고 평균을 반환합니다.
    return selected, len(eligible), best_mean


# [5] test 없이 CV 탐색 결과만 전달해 후보를 고정합니다.
selected, eligible_count, best_mean = select_candidate(searches)

print("eligible candidates:", eligible_count)
print("selected search:", selected["search"])
print("selected params:", {
    "n_estimators": int(selected["n_estimators"]),
    "max_depth": int(selected["max_depth"]),
})
print(
    "selected CV mean/std:",
    f"{selected['mean_AP']:.3f}",
    f"{selected['std_AP']:.3f}",
)

# [1] 앞 단계에서 이미 고정한 최종 하이퍼파라미터만 꺼냅니다.
# 이 아래에서는 후보를 다시 비교하거나 파라미터를 바꾸지 않습니다.
selected_params = {
    "n_estimators": int(selected["n_estimators"]),
    "max_depth": int(selected["max_depth"]),
}

# [2] 선택 후보와 같은 Random Forest를 새로 만들고,
# 후보 비교에 사용한 개발 데이터 전체로 한 번 다시 학습합니다.
# class_weight, random_state, n_jobs도 탐색 때와 동일하게 유지합니다.
final_model = RandomForestClassifier(
    **selected_params,
    class_weight="balanced",
    random_state=42,
    n_jobs=1,
)
final_model.fit(X_dev, y_dev)

# [3] 여기서부터 봉인 test를 처음으로 최종 성능 계산에 사용합니다.
# predict_proba는 클래스별 점수 열을 반환합니다.
# classes_에서 label 1의 위치를 찾아 needs_review=1 열을 명시적으로 선택합니다.
# 양성 class가 언제나 두 번째 열이라고 가정하지 않는 방식입니다.
positive_index = final_model.classes_.tolist().index(1)
test_score = final_model.predict_proba(X_test)[:, positive_index]

# AP는 고정 threshold의 0/1 예측이 아니라 양성 class 점수의 순위를 평가합니다.
test_ap = average_precision_score(y_test, test_score)

# [4] test AP는 후보 선택이 끝난 뒤 한 번 보고하는 최종 평가값입니다.
# 이 값을 본 뒤 같은 test로 후보를 다시 고르면 안 됩니다.
print("sealed test AP:", f"{test_ap:.3f}")

# 선택 후보가 사전에 정한 평균 허용폭 안에 있는지 확인합니다.
# test 봉인은 함수 입력과 위 코드의 실행 순서로 지키는 계약입니다.
assert selected["mean_AP"] >= best_mean - MEAN_TOLERANCE
```

```python
# 5-2 예제
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split

# [1] 같은 seed로 같은 모의 데이터를 다시 만들 수 있도록 난수 생성기를 고정합니다.
rng = np.random.default_rng(42)
n = 1200

# [2] 수치형 세 열과 범주형 한 열을 가진 입력 X를 만듭니다.
# prompt_tokens는 뒤에서 np.nan을 넣을 수 있도록 처음부터 float로 만듭니다.
X = pd.DataFrame({
    "prompt_tokens": rng.integers(20, 1800, n).astype(float),
    "retrieval_score": rng.normal(0.58, 0.18, n).clip(0, 1),
    "toxicity_score": rng.beta(1.5, 8.0, n),
    "route": rng.choice(
        ["chat", "rag", "agent"], n, p=[0.45, 0.40, 0.15]
    ),
})

# [3] 교육용 숨은 점수 latent를 만들고 이를 기준으로 정답 y를 만듭니다.
# 이 식은 실제 업무 규칙이 아니라 입력과 정답의 관계가 있는
# 모의 데이터를 만들기 위한 식입니다.
# 결측값을 넣기 전에 정답을 만들어 결측 여부가 정답 생성에 영향을 주지 않게 합니다.
latent = (
    -5.0
    + 3.1 * (1 - X["retrieval_score"])
    + 5.0 * X["toxicity_score"]
    + 0.0010 * X["prompt_tokens"]
    + 0.8 * (X["route"] == "agent")
    + rng.normal(0, 0.9, n)
)
y = pd.Series((latent > 0).astype(int), name="needs_review")

# [4] 다음 블록의 SimpleImputer가 실제로 작동하도록 일부 입력값을 비웁니다.
# 수치형 각 열에서 12개 셀, 범주형 route에서 8개 셀을 결측으로 만듭니다.
for column in ["prompt_tokens", "retrieval_score", "toxicity_score"]:
    missing_rows = rng.choice(n, size=12, replace=False)
    X.loc[missing_rows, column] = np.nan
X.loc[rng.choice(n, size=8, replace=False), "route"] = None

# [5] 모델 설정 선택에 사용할 개발 데이터와 마지막 평가용 test를 8:2로 나눕니다.
# stratify=y는 드문 needs_review=1 비율이 두 집합에서 크게 달라지는 것을 줄입니다.
# X_test와 y_test는 만들어 두지만 후보 선택 전에는 성능 계산에 사용하지 않습니다.
X_dev, X_test, y_dev, y_test = train_test_split(
    X, y, test_size=0.20, stratify=y, random_state=42
)

# [6] 분할 크기, 개발 데이터의 불균형 정도, 전체 입력의 결측 셀 수만 봅니다.
# 아직 test 예측이나 test 성능 계산은 하지 않습니다.
print("dev/test rows:", len(X_dev), len(X_test))
print("needs_review rate:", round(y_dev.mean(), 3))
print("missing values:", int(X.isna().sum().sum()))

from imblearn.over_sampling import RandomOverSampler
from imblearn.pipeline import Pipeline as ImbPipeline
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler

# [1] 열 이름을 유형별로 나눕니다.
# ColumnTransformer는 이 목록을 보고 각 열을 알맞은 전처리 경로로 보냅니다.
numeric_columns = [
    "prompt_tokens", "retrieval_score", "toxicity_score"
]
categorical_columns = ["route"]

# [2] 수치형 전처리 경로입니다.
# median과 평균·표준편차는 지금 계산하지 않고 fit할 때 학습 데이터에서 계산합니다.
# 결측을 먼저 채운 뒤, 단위 차이가 Logistic Regression 규제에 미치는 영향을
# 줄이도록 StandardScaler를 적용합니다.
numeric_pipeline = Pipeline([
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler()),
])

# [3] 범주형 전처리 경로입니다.
# 결측은 학습 데이터의 최빈 범주로 채우고 문자열은 One-Hot Encoding으로 바꿉니다.
# handle_unknown="ignore"는 학습 때 없던 범주를 all-zero로 바꿔
# 열 개수 오류를 막지만, 새 범주를 자동 승인하는 정책은 아닙니다.
categorical_pipeline = Pipeline([
    ("imputer", SimpleImputer(strategy="most_frequent")),
    ("onehot", OneHotEncoder(handle_unknown="ignore")),
])

# [4] 숫자형 경로와 범주형 경로를 하나의 전처리 객체로 합칩니다.
# 아직 fit하지 않았으므로 median, scaling 기준, 범주 목록은 정해지지 않았습니다.
preprocessor = ColumnTransformer([
    ("num", numeric_pipeline, numeric_columns),
    ("cat", categorical_pipeline, categorical_columns),
])

# [5] 전처리 → 불균형 처리 → 분류기 순서로 최종 Pipeline을 조립합니다.
# RandomOverSampler는 fit할 때 소수 class의 기존 행을 복제합니다.
# GridSearchCV 안에서는 각 fold의 train에만 적용되고 fold validation에는 없습니다.
# predict와 predict_proba에서는 sampler를 건너뛰므로 입력 행 수가 늘지 않습니다.
pipeline = ImbPipeline([
    ("preprocessor", preprocessor),
    ("sampler", RandomOverSampler(random_state=42)),
    ("classifier", LogisticRegression(max_iter=1000)),
])

# [6] 학습 결과가 아니라 조립된 단계의 이름과 순서만 확인합니다.
print([name for name, _ in pipeline.steps])

from sklearn.metrics import average_precision_score
from sklearn.model_selection import GridSearchCV, StratifiedKFold

# [1] 세 후보가 모두 같은 네 fold에서 비교되도록 CV 규칙을 고정합니다.
# stratified fold는 각 fold에서 needs_review=1 비율이 크게 흔들리는 것을 줄입니다.
cv = StratifiedKFold(n_splits=4, shuffle=True, random_state=42)

# [2] Pipeline 안 classifier 단계의 C를 세 값으로 비교합니다.
# "classifier__C"는 "Pipeline 단계 이름__그 단계의 파라미터" 형식입니다.
# AP는 드문 양성 class의 점수 순위를 평가합니다.
# refit=True는 최고 후보 선택 후 그 설정으로 X_dev 전체를 한 번 더 학습합니다.
search = GridSearchCV(
    pipeline,
    {"classifier__C": [0.1, 1.0, 10.0]},
    scoring="average_precision",
    cv=cv,
    n_jobs=1,
    refit=True,
)

# [3] 후보 탐색에는 개발 데이터만 전달합니다.
# 각 CV fit에서는 전처리 기준과 sampler가 해당 fold의 train에서만 학습됩니다.
# 세 후보 × 네 fold의 12회 평가 뒤, 선택 후보가 X_dev 전체에 별도로 refit됩니다.
search.fit(X_dev, y_dev)
best_pipeline = search.best_estimator_

# [4] predict_proba 열의 위치를 숫자 1로 가정하지 않고 class 순서에서 찾습니다.
# positive class는 사람 검토가 필요한 needs_review=1입니다.
classes = best_pipeline.named_steps["classifier"].classes_.tolist()
positive_index = classes.index(1)

# [5] 후보 선택과 refit이 끝난 뒤 처음으로 sealed test의 양성 class 점수를 냅니다.
# predict_proba의 출력이 실제 위험 확률로 보정됐다고 가정하지 않고
# 뒤에서는 needs_review 점수로 해석합니다.
test_score = best_pipeline.predict_proba(X_test)[:, positive_index]

# [6] CV 평가 횟수는 후보 수 × fold 수로 계산합니다.
# refit=True가 선택 후보를 X_dev 전체에 재학습한 1회는 이 숫자에 포함하지 않습니다.
candidate_count = len(search.cv_results_["params"])
cv_fit_count = candidate_count * cv.get_n_splits()

print("CV candidates/fits:", candidate_count, cv_fit_count)
print("best C:", search.best_params_["classifier__C"])
print("best CV AP:", f"{search.best_score_:.3f}")

# [7] test AP는 선택이 끝난 절차를 한 번 확인하는 최종 평가입니다.
# 이 값을 본 뒤 C나 전처리를 다시 바꾸면 test가 선택 데이터가 됩니다.
print(
    "sealed test AP:",
    f"{average_precision_score(y_test, test_score):.3f}",
)

# [8] 후보 수와 CV 평가 횟수가 강의에서 약속한 값인지 확인합니다.
assert candidate_count == 3
assert cv_fit_count == 12

# [1] 학습 때와 같은 네 열을 가진 새 요청 세 건을 만듭니다.
# 세 번째 요청의 prompt_tokens는 비어 있어 학습된 median imputer도 확인합니다.
# 새 요청에는 정답 y가 없으므로 이 단계는 평가가 아니라 inference입니다.
new_requests = pd.DataFrame({
    "prompt_tokens": [180.0, 1250.0, np.nan],
    "retrieval_score": [0.91, 0.22, 0.48],
    "toxicity_score": [0.03, 0.51, 0.17],
    "route": ["chat", "agent", "rag"],
})

# [2] best_pipeline은 refit=True에 의해 X_dev 전체로 이미 학습됐습니다.
# 학습된 imputer·scaler·encoder가 새 입력을 변환한 뒤 classifier가 점수를 냅니다.
# positive_index를 사용해 needs_review=1에 해당하는 열만 선택합니다.
before_score = best_pipeline.predict_proba(
    new_requests
)[:, positive_index]

# [3] predict는 별도 업무 threshold가 아니라 classifier의 기본 규칙으로 0/1을 냅니다.
# inference에서는 RandomOverSampler가 실행되지 않으므로 세 요청은 세 예측입니다.
before_prediction = best_pipeline.predict(new_requests)

# [4] 점수와 예측을 한 표로 묶습니다.
# 두 배열은 다음 블록에서 저장·reload 결과를 비교할 기준값으로도 사용합니다.
result = pd.DataFrame({
    "needs_review_score": before_score,
    "prediction": before_prediction,
})

# [5] 화면에서는 점수를 소수점 셋째 자리까지 표시하지만 원래 배열 값은 바꾸지 않습니다.
print(result.to_string(
    index=False,
    formatters={"needs_review_score": "{:.3f}".format},
))

import joblib

# [1] 선택과 refit이 끝난 best_pipeline을 현재 노트북 작업 폴더에 저장합니다.
# imputer·scaler·encoder·sampler 설정·classifier의 학습 상태가 한 파일에 들어갑니다.
joblib.dump(best_pipeline, "review_pipeline.joblib")

# [2] joblib은 load 중 코드를 실행할 수 있으므로

loaded_classes = loaded_pipeline.named_steps[

loaded_positive_index = loaded_classes.index(1)

score_close = np.allclose(before_score, after_score)
prediction_match = np.array_equal(

print("reload scores close:", score_close)

# [6] 하나라도 다르면 셀을 실패시켜 눈으로 True를 놓치는 일을 막습니다.
```

## 헷갈렸던 부분 & 해결
- 튜터님이 하이퍼파라미터 얘기 하시면서 Grid에 대해서 얘기하시다가 탐욕이라고 언급하셨는데, 잘못된 언급이었기 때문에 약간 혼동이 왔다.
- 탐욕 알고리즘의 그리드는 greed이고, grid는 격자에 해당한다.
- 또한 기본적으로 탐욕 알고리즘은 매 반복에서 최선을 선택해나가는 알고리즘이기 때문에 완전히 잘못된 것이었다.
- 이전 실습에서도 나왔었지만 imblearn라이브러리의 파이프라인에 대해서 뭔가 약간 다르기에 헷갈렸다.
- 클로드를 통해서 확인해본 결과 기본적인 사이킷런의 파이프라인 같은 경우에는 기본적으로 입력 행렬과 출력 행렬의 크기가 변해서는 안된다는 틀이 있었다.
- 그러나 샘플링을 하기 위해서는 행의 길이가 변동이 있어야 했기 때문에 `SMOTE`나 `RandomOverSampler`는 기본 사이킷런의 파이프라인에서 사용할 수 없었다.
- 그래서 나온 것이 imbalanced-learn이다.
- imblearn은 불균형 데이터처리에 특화된 사이킷런 호환 라이브러리이기 때문에 가능한 처리였던 것이었다.
