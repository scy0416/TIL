## 오늘 배운 것
- 파인튜닝의 채점을 위해서는 일단 파일의 지문이 같은지부터 확인해야 한다.
- 서로 다른 파일을 사용해서 진행했다면 그 결과들은 서로 비교할 수 있는 것들이 아니다.
- 성능을 평가하기 위해서는 생성 원문을 편집하지 않고 그대로 저장하고서 다음과 같은 검사들을 한다.
  - 형식 검사: JSON 형식 및 타입, 필드 검사
  - 업무 의미: 형식 검사를 통과한 경우 진행하며, 업무적 의미에 맞게 답했는지를 확인한다.
  - 참조 일치: 모범 답안 객체와 같은지를 검사한다.
- 형식 검사는 json객체 여부 검사, 스키마 검사, citation 검사 순서로 진행한다.
- 형식 검사는 의미적이 아니라 형식적인 검사를 진행하기 때문에 의미적인 오답을 걸러내지를 못한다.
- 업무 의미적인 오답은 종류가 많기 때문에 검사할 의미 유형을 분리해야 한다.
- 답안에 대한 비교를 하기 위해서는 테스트 조건을 전부 맞춰야 한다.
- 기본적으로 L4인지 T4인지도 확인하고서 맞춰줘야 한다.

## 왜 중요한가
- 파인튜닝을 진행한 모델에 대한 성능을 평가하는 부분이기도 하며 서비스에 직접 연결하는 부분이기 때문에 중요하다.
- 모델을 파인튜닝하는 것도 중요하지만 그걸 사용할 수 있게 만드는 것도 중요하다.

```python
# 형식 검사 예제
# 형식이 올바른 오답도 통과하는 예제로 자동 검사의 범위를 확인합니다.
def check_format_examples(wrong_raw):
    assert all(
        inspect_response(wrong_raw, {"policy01-v1"}).values()
    ), '형식이 올바른 예제는 형식 검사를 통과해야 합니다.'
    assert not inspect_response(
        "```json\n" + wrong_raw + "\n```", {"policy01-v1"}
    )["json_object"], (
        'JSON 객체가 아닌 원문을 검사에서 거절해야 합니다.'
    )
    assert not inspect_response(
        '{"status":[],"answer":"답변","citations":[]}', set()
    )["schema_ok"], (
        '잘못된 필드나 자료형을 형식 검사에서 거절해야 합니다.'
    )
    assert not inspect_response(
        '{"status":"abstained","answer":"","citations":[]}', set()
    )["schema_ok"], (
        '잘못된 필드나 자료형을 형식 검사에서 거절해야 합니다.'
    )
    assert not inspect_response(wrong_raw, {"policy01-v2"})[
        "citation_ids_ok"
    ], '제공 문서에 없는 인용 ID를 검사해야 합니다.'


def inspect_response(raw, source_ids):
    # JSON 앞뒤 설명과 코드 펜스를 제거하지 않습니다. 원문이 계약을 지켰는지 확인합니다.
    result = {
        "json_object": False,
        "schema_ok": False,
        "citation_ids_ok": False,
    }
    try:
        value = json.loads(raw)
    except json.JSONDecodeError:
        return result
    if not isinstance(value, dict):
        return result
    result["json_object"] = True
    status, answer, citations = (
        value.get("status"),
        value.get("answer"),
        value.get("citations"),
    )
    valid_fields = (
        set(value) == {"status", "answer", "citations"}
        and isinstance(status, str)
        and status in {"answered", "abstained"}
        and isinstance(answer, str)
        and bool(answer.strip())
        and isinstance(citations, list)
        and all(isinstance(x, str) for x in citations)
    )
    if not valid_fields:
        return result
    if (status == "answered" and not citations) or (
        status == "abstained" and citations
    ):
        return result
    result["schema_ok"] = True
    result["citation_ids_ok"] = set(citations).issubset(source_ids)
    return result


# 설명용 반례입니다. 문장 내용은 잘못됐지만 JSON, 필드와 문서 ID는 모두 맞습니다.
wrong_meaning = {
    "status": "answered",
    "answer": "70,000원은 65,000원 이하이므로 사전 승인이 필요하지 않습니다.",
    "citations": ["policy01-v1"],
}
wrong_raw = json.dumps(wrong_meaning, ensure_ascii=False)
check_format_examples(wrong_raw)
print(
    "형식이 정상인 의미 오류:",
    inspect_response(wrong_raw, {"policy01-v1"}),
)
print(wrong_raw)
```

```python
# 코드로 진행할 수 있는 단계별 검사의 결과를 확인하는 예제
# 모든 응답을 분모에 남기고 JSON, 필드, 인용 검사 단계별 통과 수를 셉니다.
summary = []
for name in comparison:
    rows = [item for item in responses if item["model"] == name]
    counts = {
        key: sum(item["format_checks"][key] for item in rows)
        for key in ["json_object", "schema_ok", "citation_ids_ok"]
    }
    counts.update(
        model=name,
        total=len(rows),
        format_ok=sum(
            all(item["format_checks"].values()) for item in rows
        ),
    )
    summary.append(counts)
    print(name, counts)

# 실패 단계는 첫 실패 한 개만 셉니다. 뒤 검사를 생략한 False를 별도 실패 세 개로 부풀리지 않습니다.
failure_counts = {}
for name in comparison:
    stages = Counter()
    for item in [x for x in responses if x["model"] == name]:
        stage = next(
            (
                key
                for key, passed in item["format_checks"].items()
                if not passed
            ),
            "format_passed",
        )
        stages[stage] += 1
    failure_counts[name] = dict(stages)
    assert (
        sum(stages.values()) == 12
    ), '첫 실패 단계의 합계가 dev 12개와 다릅니다.'
    print(name, "첫 실패 단계:", dict(stages))
print(
    "모든 비율의 분모는 dev 12개입니다. 형식 통과 수는 업무 정답 수가 아닙니다."
)
```

```python
# 파인튜닝한 모델을 api서버에 연결하는 예제
# 입력 검사, 검색, 실제 모델 생성, 출력 검사 순서로 요청을 처리합니다.
class AnswerRequest(BaseModel):
    # 숫자를 문자열로 바꾸거나 여분 필드를 묵인하지 않습니다. 이는 의미 판단이 아닌 API 입력 검사입니다.
    model_config = ConfigDict(
        strict=True, extra="forbid", str_strip_whitespace=True
    )
    team: str = Field(min_length=1, max_length=20)
    question: str = Field(min_length=1, max_length=500)


app = FastAPI(title="출장 규정 SFT 연결 예제")


@app.post("/answer")
def answer(request: AnswerRequest):
    documents = retrieve_documents(request.question, request.team)
    try:
        generated = generate_answer(request.question, documents)
    except ValueError as error:
        return JSONResponse(
            status_code=422, content={"detail": str(error)}
        )

    # API가 반환 가능한 형식인지 검사하며, 답의 업무상 정답 여부는 별도로 검토합니다.
    checks = inspect_response(
        generated["raw_response"], {doc["id"] for doc in documents}
    )
    format_ok = all(checks.values()) and not generated["length_limited"]
    body = {
        **generated,
        "retrieved_documents": documents,
        "format_checks": checks,
        "response": (
            json.loads(generated["raw_response"]) if format_ok else None
        ),
        "review_required": True,
    }

    # 형식 실패는 응답을 고쳐 정상으로 꾸미지 않습니다. 200도 업무 의미 승인이라는 뜻은 아닙니다.
    return JSONResponse(
        status_code=200 if format_ok else 502, content=body
    )


api_examples = []

        assert result.status_code in {

        }, 'API의 정상 응답, 형식 실패 또는 입력 거절 상태를 확인하세요.'
        print(json.dumps(record, ensure_ascii=False, indent=2))

    # 잘못된 입력은 모델을 호출하기 전에 422로 거절합니다.
    for bad in [
        {"team": "가온", "question": 123},
```

## 헷갈렸던 부분 & 해결
- 교안에서 테스트의 환경을 맞추는 것나 지문을 확인하는 것을 보고서 manifest랑 무슨 관계이고, 어떻게 다른건지 궁금해졌다.
- 확인해보니 지문은 sha256처럼 문서의 동일성을 확인할 수 있는 정보이고, manifest는 그 지문 정보를 포함해서 결과를 이해하고 검증하고 재현하는 데 필요한 정보들을 포함한 명세서라고 한다.
- 그리고 문득 citations에 대해서 헷갈려서 이에 대해서 조금 살펴봤다.
- 챗봇은 기본적으로 검색된 문서인 sources를 사용해서 답변을 하며, 이 sources에 정보가 충분한지 아닌지에 따라서 답변을 할 수도 있고(answer), 보류할 수도(abstain) 있다.
- 이 때, 답변을 한다면 LLM은 제공받은 문서들 중에서 사용한 문서를 citations로 만들어서 반환해야 하며, 이 때, 모든 sources를 사용하지 않아도 괜찮지만 sources에 없는 문서 또는 청크를 사용했다고 주장해서는 안된다.
- 만약 답변을 보류하는 상황에서는 답변을 하지 않았기 때문에 citations에 아무런 값이 있어서는 안된다.
