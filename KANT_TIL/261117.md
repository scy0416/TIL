## 오늘 배운 것
- 학습용 데이터는 강의에서는 규칙 함수로 진행하지만 실제는 LLM으로 초안을 만들고 사람 검수나 코드 검수를 진행한다고 한다.
- `add_generation_prompt=True`는 무조건 출력 프롬프트를 추가하지만 `continue_final_message=True`는 출력의 중간부터 이어쓰도록 만들 수 있다.
- `return_dict`는 입력 토큰id와 어텐션 마스크까지 받는데, 이건 실제 모델의 입력을 할 때 필요하기 때문에 토큰화 관련 사용시에는 불필요하다.
- 단순 얼라인먼트에서의 학습 데이터셋은 1000개로도 가능하다고 한다.
- 만들어진 학습용 데이터는 다섯 단계로 레코드를 검사한다.
  - 식별 정보
  - 대화
  - 응답 형식
  - 인용
  - 업무 규칙
- 검사를 위한 함수를 만들었으면 일부러 틀린 데이터를 넣어서 문제 있는 데이터를 의도대로 잡아내는지 확인한다.
- 만들어진 학습용 데이터를 분할할 때에는 모든 데이터를 무작위로 섞지 않고 그룹별로 섞어서 골라야 한다.
- 만약 그냥 전부 섞어버리면 데이터 누수가 발생해서 결과를 신뢰하기 어려워질 수 있다.
- 여기서도 마찬가지로 test는 최종 딱 한 번만 확인한다.
- 학습용 데이터셋을 만들면 관련된 파일을 만들어서 관리한다.
  - train.jsonl: 만들어진 데이터셋-train
  - dev.jsonl: 만들어진 데이터셋-dev
  - test.jsonl: 만들어진 데이터셋-test
  - dataset_info.json: 데이터가드
  - training_data.zip: 위를 전부 포함한 압축파일
- dataset_info의 필드
  - synthetic, limitations: 무엇이고, 한계는
  - model_id, revision, system_prompt: 같은 템플릿, 지시
  - split_seed, split_groups: 분할 재현
  - max_length, max_observed: 512(최대 토큰 길이), 380(관측된 토큰 중 가장 긴 길이)
  - file_sha256: 무결성

## 왜 중요한가
- 파인튜닝을 위한 데이터셋을 만들고, 관리하는 전반적인 내용에 대한 것이기 때문에 중요한다.

```python
# 데이터셋 생성 예제
# 같은 규정에서 파생된 사례가 나중에 서로 다른 집합으로 흩어지지 않게 묶습니다.
# 각 부서는 독립된 가상 규정 계열입니다. 구판과 개정판은 같은 group_id를 사용합니다.
teams = [
    "가온",
    "나래",
    "다온",
    "라온",
    "마루",
    "바른",
    "새봄",
    "아람",
    "온유",
    "이룸",
    "자람",
    "하람",
]
limits = [
    65000,
    75000,
    85000,
    95000,
    105000,
    115000,
    70000,
    80000,
    90000,
    100000,
    110000,
    120000,
]
groups = [
    {
        "group_id": f"policy{i:02d}",
        "team": team,
        "limit": limit,
        "updated_limit": limit + 10000,
    }
    for i, (team, limit) in enumerate(zip(teams, limits), start=1)
]


# 개정 여부에 따라 한도와 문서 ID를 함께 바꿔 구판과 혼동하지 않게 합니다.
def make_document(group, updated=False):
    limit = group["updated_limit"] if updated else group["limit"]
    version = "v2" if updated else "v1"
    return {
        "id": f'{group["group_id"]}-{version}',
        "text": f'{group["team"]}팀의 현재 숙박비 지원 한도는 1박 {limit:,}원입니다. '
        "한도 이하의 예약에는 사전 승인이 필요하지 않습니다. "
        "한도를 초과하는 예약은 예약 전 서면 승인을 받아야 합니다.",
    }


case_specs = []
for group in groups:
    old_document = make_document(group)
    new_document = make_document(group, updated=True)

    # 같은 질문 금액을 구판과 개정판에 적용하는 사례도 포함합니다.
    for kind, amount, document in [
        ("below", group["limit"] - 5000, old_document),
        ("boundary", group["limit"], old_document),
        ("over_limit", group["limit"] + 5000, old_document),
        ("updated", group["limit"] + 5000, new_document),
        ("missing", None, None),
        ("irrelevant", None, old_document),
    ]:
        # 그룹명은 질문의 실제 문맥이고 group_id는 분리용 메타데이터입니다.
        question = (
            f'{group["team"]}팀 출장에서 1박 {amount:,}원인 방을 예약하려면 무엇을 해야 하나요?'
            if amount is not None
            else f'{group["team"]}팀 출장 중 통신비는 얼마까지 지원되나요?'
        )
        case_specs.append(
            {
                "id": f'{group["group_id"]}-{kind}',
                "group_id": group["group_id"],
                "case_type": kind,
                "question": question,
                "documents": [] if document is None else [document],
                "amount": amount,
                "limit": (
                    group["updated_limit"]
                    if kind == "updated"
                    else group["limit"]
                ),
            }
        )

print("규정 그룹:", len(groups), "사례:", len(case_specs))
print("유형별 수:", dict(Counter(s["case_type"] for s in case_specs)))

# 문서의 한도와 질문 금액을 비교해 학습에 사용할 모범답안을 만듭니다.
def reference_answer(spec):
    # 이 함수는 수업용 규칙의 모범답안을 만듭니다. 모델이 생성한 답을 정답으로 복사하지 않습니다.
    if spec["case_type"] in {"missing", "irrelevant"}:
        return {
            "status": "abstained",
            "answer": "제공된 근거로는 출장 중 통신비 지원 한도를 확인할 수 없습니다. 담당자에게 확인하세요.",
            "citations": [],
        }
    amount, limit = spec["amount"], spec["limit"]
    if amount > limit:
        answer = (
            f"숙박비 지원 한도는 1박 {limit:,}원입니다. "
            f"질문의 1박 {amount:,}원은 한도를 초과하므로 예약 전 서면 승인을 받아야 합니다."
        )
    else:
        relation = (
            "한도와 같으므로" if amount == limit else "한도 이하이므로"
        )
        answer = (
            f"숙박비 지원 한도는 1박 {limit:,}원입니다. "
            f"질문의 1박 {amount:,}원은 {relation} 사전 승인이 필요하지 않습니다."
        )
    return {
        "status": "answered",
        "answer": answer,
        "citations": [spec["documents"][0]["id"]],
    }


# 1장의 85,000원 규정을 포함한 가상 계열에서 초과와 개정을 나란히 읽습니다.
for spec in case_specs:
    if spec["group_id"] == "policy03" and spec["case_type"] in {
        "over_limit",
        "updated",
    }:
        print(spec["id"], spec["question"])
        print(json.dumps(reference_answer(spec), ensure_ascii=False))
```

```python
# 데이터셋 검사 예제
# 한 레코드를 식별 정보, 대화, 응답 형식, 업무 조건 순서로 확인합니다.
def check_message_fields(messages):
    assert [m["role"] for m in messages] == [
        "system",
        "user",
        "assistant",
    ], '대화 역할은 system, user, assistant 순서여야 합니다.'
    assert all(
        set(m) == {"role", "content"} and isinstance(m["content"], str)
        for m in messages
    ), '대화는 role과 문자열 content로 구성해야 합니다.'
    assert (
        messages[0]["content"] == SYSTEM_PROMPT
    ), 'system 지시가 이 수업의 출력 계약과 다릅니다.'


def check_response_fields(response):
    assert isinstance(response, dict) and set(response) == {
        "status",
        "answer",
        "citations",
    }, '모범답안은 status, answer, citations를 가진 JSON 객체여야 합니다.'
    assert response["status"] in {
        "answered",
        "abstained",
    }, 'status는 answered 또는 abstained여야 합니다.'
    assert (
        isinstance(response["answer"], str)
        and response["answer"].strip()
    ), 'answer에는 비어 있지 않은 문자열이 필요합니다.'


def check_reference_rule(spec, response, citations):
    # 업무 조건: 근거 부족, 한도 초과, 한도 이하를 각각 확인합니다.
    answer = response["answer"]
    if spec["case_type"] in {"missing", "irrelevant"}:
        assert (
            response["status"] == "abstained" and citations == []
        ), '근거 부족 사례는 abstained와 빈 인용 목록이어야 합니다.'
        assert (
            "확인할 수 없습니다" in answer and "담당자" in answer
        ), '근거 부족 이유와 담당자 확인 안내가 필요합니다.'
    else:
        assert (
            response["status"] == "answered" and len(citations) == 1
        ), '답할 수 있는 사례에는 answered와 근거 ID 하나가 필요합니다.'
        assert (
            f'1박 {spec["limit"]:,}원' in answer
            and f'1박 {spec["amount"]:,}원' in answer
        ), '질문 금액과 1박 한도를 모두 보존해야 합니다.'
        if spec["amount"] > spec["limit"]:
            assert (
                "초과" in answer
                and "예약 전 서면 승인을 받아야" in answer
            ), '한도 초과와 예약 전 서면 승인 필요를 설명해야 합니다.'
            assert (
                "사전 승인이 필요하지" not in answer
            ), '초과 사례에 승인 불필요라는 표현이 들어갔습니다.'
        else:
            assert (
                "사전 승인이 필요하지 않습니다" in answer
            ), '한도 이하 사례에는 사전 승인 불필요를 설명해야 합니다.'
            assert (
                "예약 전 서면 승인을 받아야" not in answer
            ), '한도 이하 사례를 사전 승인 필요로 바꾸면 안 됩니다.'


specs_by_id = {spec["id"]: spec for spec in case_specs}


def validate_record(row):
    # 식별 정보: 레코드가 원래 사례 명세와 연결되는지 확인합니다.
    assert set(row) == {
        "id",
        "group_id",
        "case_type",
        "messages",
    }, '레코드에는 id, group_id, case_type, messages만 있어야 합니다.'
    spec = specs_by_id[row["id"]]
    assert (
        row["group_id"] == spec["group_id"]
        and row["case_type"] == spec["case_type"]
    ), '규정 그룹이나 사례 유형이 원래 명세와 다릅니다.'

    # 대화: 역할 순서와 문자열 형식을 확인한 뒤 질문과 근거를 대조합니다.
    messages = row["messages"]
    check_message_fields(messages)
    user = json.loads(messages[1]["content"])
    assert user == {
        "question": spec["question"],
        "documents": spec["documents"],
    }, '질문이나 근거 문서가 원래 사례 명세와 다릅니다.'

    # 응답 형식: 필요한 세 필드와 자료형을 확인합니다.
    response = json.loads(messages[2]["content"])
    check_response_fields(response)

    # 인용: 제공 문서에 없는 ID와 중복 ID를 거절합니다.
    citations = response["citations"]
    assert isinstance(citations, list) and all(
        isinstance(x, str) for x in citations
    ), 'citations에는 문서 ID 문자열 목록이 필요합니다.'

    # 제공 문서 ID는 인용 허용 범위이며, 답변의 의미가 맞다는 증거는 아닙니다.
    source_ids = {doc["id"] for doc in user["documents"]}
    assert set(citations) <= source_ids and len(citations) == len(
        set(citations)
    ), '인용 ID가 제공 문서에 없거나 중복되었습니다.'

    # shortcut: 이 교재의 정형 문구만 확인합니다. 다른 문장과 실제 규정에는 별도 의미 검토가 필요합니다.
    check_reference_rule(spec, response, citations)


# 검사를 모두 통과한 원본만 다음 자료 분리 단계에서 사용합니다.
for row in rows:
    validate_record(row)
assert (
    len({r["id"] for r in rows}) == len(rows) == 72
), '중복 ID 없이 원본 72개 사례가 있어야 합니다.'
print("형식과 교재 규칙 검사:", len(rows), "/", len(rows), "통과")
print(
    "응답 상태:",
    dict(
        Counter(
            json.loads(r["messages"][2]["content"])["status"]
            for r in rows
        )
    ),
)
```

```python
# 데이터셋 분할 예제
# 행이 아닌 규정 그룹을 나누어 같은 규정의 구판과 개정판이 함께 이동하게 합니다.
def check_split_overlap(split_groups, splits, groups):
    for left, right in [
        ("train", "dev"),
        ("train", "test"),
        ("dev", "test"),
    ]:
        assert split_groups[left].isdisjoint(
            split_groups[right]
        ), '규정 그룹이 두 집합에 동시에 들어갔습니다.'
        left_inputs = {
            r["messages"][1]["content"] for r in splits[left]
        }
        right_inputs = {
            r["messages"][1]["content"] for r in splits[right]
        }
        assert left_inputs.isdisjoint(
            right_inputs
        ), '같은 사용자 입력이 두 집합에 중복되었습니다.'
    assert set.union(*split_groups.values()) == {
        g["group_id"] for g in groups
    }, '분리 과정에서 규정 그룹이 빠지거나 추가되었습니다.'
    assert {name: len(part) for name, part in splits.items()} == {
        "train": 48,
        "dev": 12,
        "test": 12,
    }, 'train/dev/test의 행 수는 48/12/12여야 합니다.'


# 섞기 전 그룹 순서를 고정해야 같은 seed에서 같은 분할을 얻습니다.
group_ids = sorted({row["group_id"] for row in rows})
random.Random(SEED).shuffle(group_ids)
split_groups = {
    "train": set(group_ids[:8]),
    "dev": set(group_ids[8:10]),
    "test": set(group_ids[10:]),
}
splits = {
    name: [row for row in rows if row["group_id"] in selected_groups]
    for name, selected_groups in split_groups.items()
}

# 행을 섞어 자르는 대신 그룹을 먼저 나눕니다. 같은 계열의 여섯 행은 함께 움직입니다.
check_split_overlap(split_groups, splits, groups)
for name, part in splits.items():
    print(name, "행:", len(part), "그룹:", sorted(split_groups[name]))
    print("유형:", dict(Counter(row["case_type"] for row in part)))
DEV_PROBE_ID = next(
    r["id"] for r in splits["dev"] if r["case_type"] == "over_limit"
)
print("뒤 장의 공통 dev 문항:", DEV_PROBE_ID)
```

```python
# 데이터셋 파일 저장 예제
# 뒤 장에서 다시 만들지 않고 읽을 수 있도록 같은 자료를 JSONL과 ZIP으로 보관합니다.
# JSONL은 한 줄이 레코드 한 개입니다. 파일과 ZIP 모두 같은 이름으로 다시 실행하면 덮어씁니다.
output_dir = Path("training_data")
output_dir.mkdir(exist_ok=True)
file_hashes = {}
for name, part in splits.items():
    path = output_dir / f"{name}.jsonl"
    payload = "".join(
        json.dumps(row, ensure_ascii=False) + "\n" for row in part
    )
    path.write_text(payload, encoding="utf-8")
    file_hashes[path.name] = hashlib.sha256(
        path.read_bytes()
    ).hexdigest()

dataset_info = {
    "dataset_id": DATASET_ID,
    "synthetic": True,
    "model_id": MODEL_ID,
    "revision": REVISION,
    "system_prompt": SYSTEM_PROMPT,
    "record_fields": ["id", "group_id", "case_type", "messages"],
    "response_fields": ["status", "answer", "citations"],
    "split_seed": SEED,
    "split_groups": {
        name: sorted(ids) for name, ids in split_groups.items()
    },
    "split_counts": {name: len(part) for name, part in splits.items()},
    "groups": groups,
    "max_length": MAX_LENGTH,
    "max_observed_tokens": max(lengths.values()),
    "dev_probe_id": DEV_PROBE_ID,
    "file_sha256": file_hashes,
    "runtime": runtime,
    "limitations": "규칙과 문장 틀을 공유한 합성 교재 자료이며 실제 업무 일반화 성능을 보장하지 않습니다.",
}
(output_dir / "dataset_info.json").write_text(
    json.dumps(dataset_info, ensure_ascii=False, indent=2),
    encoding="utf-8",
)
archive_path = Path("training_data.zip")
with zipfile.ZipFile(
    archive_path, "w", compression=zipfile.ZIP_DEFLATED
) as archive:
    for name in [
        "train.jsonl",
        "dev.jsonl",
        "test.jsonl",
        "dataset_info.json",
    ]:
        archive.write(output_dir / name, arcname=name)

# ZIP 안의 행을 다시 읽어 문자 인코딩과 행 수까지 확인합니다.
with zipfile.ZipFile(archive_path) as archive:
    print("ZIP 파일:", archive.namelist())
    for name in splits:
        restored = [
            json.loads(line)
            for line in archive.read(f"{name}.jsonl")
            .decode("utf-8")
            .splitlines()
        ]
        assert (
            restored == splits[name]
        ), 'ZIP에서 다시 읽은 자료가 저장 전 자료와 다릅니다.'
print("저장 위치:", archive_path.resolve())
print("자료:", dataset_info["split_counts"])
```
