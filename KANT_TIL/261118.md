## 오늘 배운 것
- FP32랑 다르게 BF16은 메모리 효율이 좋고 표현력은 비슷한데, 소수점 정밀도를 포기한다.
- BF16은 구글이 만든 brain float로 표준처럼 사용된다.
- T4는 오래된거라 기본적으로 지원하지 않지만 L4는 기본 지원된다고 한다.
- max_steps, per_device_train_batch_size, gradient_accumulation_steps는 같이 생각해야 한다.
- 하나의 배치는 per_device_train_batch_size, gradient_accumulation_steps를 곱한 값만큼이 된다.
- per_device_train_batch_size는 그래프카드가 2장이면 또 곱해진다.
- 배치 사이즈 곱하기 max_steps가 몇 번 처리할 것인지에 대한 정보이고, 이를 통해서 몇 epoch를 돌지 알 수 있다.
- lr_scheduler_type, warmup_steps는 두 갱신동안 올리고 이후에는 천천히 감소하는데, 그냥 잘되니까 그렇게 한다고 한다.
- lr_scheduler_type은 lr의 감소방식을 선택하고, warmup은 최대 lr까지 증가하는 시간을 정한다.
- adamw가 표준처럼 사용된다고 한다.
  - adamw의 w는 weight_decay의 w이다.
- weight_decay가 규제값-가중치 값이 너무 커지는걸 막아주는 역할을 한다.
  - L2정규화에 해당한다.
- sdpa: scaled dot product attention
  - pytorch내장 어텐션 구현을 의미한다.
- numel은 가중치 개수를 반환한다.
- wandb(w&b)는 머신러닝 실험을 기록하고 시각화해주는 서비스이다.
- 파인튜닝을 했을 때, loss가 작아진다고 하더라도 의미적으로 제대로 대답한다는걸 뜻하지는 않는다.
  - 출력 형식을 제어하는데에는 효과가 좋다.
- 학습된 모델의 정보를 저장할 때에는 다음 항목들을 저장한다.
  - `chat_template.jinja`
  - `config.json`
  - `generation_config.json`
  - `model.safetensors`
  - `tokenizer.json`
  - `tokenizer_config.json`

```python
# 학습용 라벨 데이터를 인코딩하는 함수
def encode_example(row):
    messages = row["messages"]

    # 정답이 없는 프롬프트와 정답까지 포함한 대화의 경계를 비교합니다.
    prompt_ids = tokenizer.apply_chat_template(
        messages[:2],
        tokenize=True,
        add_generation_prompt=True,
        return_dict=False,
    )
    full_ids = tokenizer.apply_chat_template(
        messages,
        tokenize=True,
        add_generation_prompt=False,
        return_dict=False,
    )
    if full_ids[: len(prompt_ids)] != prompt_ids:
        raise ValueError(
            f"프롬프트 경계가 일치하지 않습니다: {row['id']}"
        )
    answer_ids = full_ids[len(prompt_ids) :]
    if answer_ids.count(tokenizer.eos_token_id) != 1:
        raise ValueError(
            f"답변 종료 토큰을 하나로 확인할 수 없습니다: {row['id']}"
        )
    end = len(prompt_ids) + answer_ids.index(tokenizer.eos_token_id) + 1
    input_ids = full_ids[:end]

    # 길이를 몰래 자르면 답변이나 종료 신호가 사라집니다. 원인을 고친 뒤 다시 준비합니다.
    if len(input_ids) > MAX_LENGTH:
        raise ValueError(
            f"{row['id']}: {len(input_ids)} tokens > MAX_LENGTH={MAX_LENGTH}"
        )

    # -100은 손실에서 제외하는 표시입니다. input_ids의 질문과 근거는 그대로 둡니다.
    # 프롬프트는 문맥으로 남기고, 정답을 맞히는 손실에서만 제외합니다.
    labels = [-100] * len(prompt_ids) + input_ids[len(prompt_ids) :]
    assert labels[-1] == tokenizer.eos_token_id and any(
        value != -100 for value in labels
    ), '손실 대상에는 답변과 종료 토큰이 포함되어야 합니다.'
    return {
        "input_ids": input_ids,
        "attention_mask": [1] * len(input_ids),
        "labels": labels,
    }
```

```python
def collate(features):
    # 패딩 ID와 종료 ID가 우연히 같더라도 위치를 기준으로 labels를 패딩합니다.
    return {
        "input_ids": pad_sequence(
            [torch.tensor(item["input_ids"]) for item in features],
            batch_first=True,
            padding_value=tokenizer.pad_token_id,
        ),
        "attention_mask": pad_sequence(
            [torch.tensor(item["attention_mask"]) for item in features],
            batch_first=True,
            padding_value=0,
        ),
        "labels": pad_sequence(
            [torch.tensor(item["labels"]) for item in features],
            batch_first=True,
            padding_value=-100,
        ),
    }
```

```python
# 학습한 결과 모델을 이용한 추론을 하기 위한 함수
def generate_probe(model, row):
    model.eval()

    # assistant 모범답안을 제외해야 학습 전 실제 생성 능력을 관찰할 수 있습니다.
    inputs = tokenizer.apply_chat_template(
        row["messages"][:2],
        tokenize=True,
        add_generation_prompt=True,
        return_dict=True,
        return_tensors="pt",
    ).to(model.device)

    # 이 구간의 연산은 역전파 그래프로 기록하지 않습니다.
    with torch.inference_mode():
        output = model.generate(
            **inputs,
            generation_config=generation,
        )

    # 생성 결과에 포함된 입력 부분을 제외하고 새 답변 토큰만 읽습니다.
    new_tokens = output[0, inputs["input_ids"].shape[1] :]
    eos_ids = (
        generation.eos_token_id
        if isinstance(generation.eos_token_id, list)
        else [generation.eos_token_id]
    )
    ended = new_tokens[-1].item() in eos_ids
    return {
        "case_id": row["id"],
        "group_id": row["group_id"],
        "case_type": row["case_type"],
        "messages": row["messages"][:2],
        "raw_response": tokenizer.decode(
            new_tokens, skip_special_tokens=True
        ),
        "input_tokens": inputs["input_ids"].shape[1],
        "new_tokens": new_tokens.numel(),
        "ended_with_eos": ended,
        "length_limited": new_tokens.numel()
        == generation.max_new_tokens
        and not ended,
    }
```

## 헷갈렸던 부분 & 해결
- toy_logits를 통해서 텐서 사이의 loss를 한 칸 밀어서 계산한다는 설명을 보고서 한 번 쭉 만든 다음에 비교한다고 잘못생각했다.
- 튜터님께 질문한 결과 결국에는 일반적인 방식의 학습과 크게 다를 바가 없었고, 단지 마지막의 assistant부분에 대해서만 추가적인 학습을 진행할 뿐이었다.
- 교안에서는 싱글턴에 대해서만 진행했지만 멀티턴도 충분히 가능하지 않을까 하는 의문이 생겨 튜터님께 질문을 했다.
- 튜터님께서는 멀티턴에 대해서도 진행할 수 있지만 생각만큼 결과가 좋지 않을거라고 하셨고, 이는 튜터님이 체득하신 영역이다.
- 확인한 결과 일반 Trainer가 아니라 SFTTrainer를 통하면 멀티턴에 대해서도 가능하고, 심지어 여기에서 사용하는 방법은 마지막 assistant메시지만 학습하는게 아니라 모든 Assistant메시지들을 학습하는 방식으로 진행한다.
- 마지막 assistant에 대해서만 생각했지 모든 assistant에 대해서 학습하는 방법은 생각을 못했는데, 확실히 그쪽이 효과가 좋을거 같다.
- 그리고 설명해주시다가 멀티턴으로 학습시키기보다는 과거 내용을 압축 또는 요약해서 포함하는 얘기에 대해서 나왔는데, 이에 대해서 요약하는 모델을 따로 두어 요약한 결과를 이용했을 때 결과가 좋게 나오도록 하기 위해 원래 LLM의 가중치가 아니라 요약 모델의 가중치를 학습하는 방법이 유효할지에 대해서 질문했다.
- 이미 그쪽으로 연구하는 사람이 있을거라고 했고, claude도 그럴거라고 했다.
