- 파이썬 함수의 매개변수에서 중간에 *이 들어가면 그 다음부터는 무조건 키워드로 넣어줘야한다
- SLI는 Service Level Indicators로, 시스템 가용성을 파악하기 위한 핵심 측정치와 지표를 의미한다.
- 로그가 있더라도 상태에 따라서 지표마다 모든 로그를 사용하지 않을 수 있다.
- 비용이 들지 않은 경우에는 비용상정에 사용하지 않고, 지연이 없던 경우나 측정이 제대로 되지 않은 경우에도 사용하지 않을 수 있다.
- 대표적으로는 완료, 명시적 실패, 미확인이 있다.
- p50은 전체 중에서 50%에 해당하는 값이고, p95는 전체 중 95%에 해당하는 값이다.
- 확인되지 않은 값은 특정한 값으로 유추하지 않고 확인되지 않음으로 둔다.
  - 특정한 값이나 실패로 처리하면 지표가 오염된다.
- LLM사용에서 문맥의 길이가 길어지면 토큰 사용량과 비용은 비례해서 늘어나는게 당연하지만 지연 시간은 경향성이 있을 뿐 확신할 수는 없다.
- 지표상으로 측정할 수 없는 질적인 문제는 사용자 피드백을 받아야만 한다.
- 피드백 응답률은 전체에 대한 응답수, 긍정 비율은 전체 피드백 중 긍정 피드백 수이다.
- langfuse에서 비용은 generation에 대해서만 확인한다.
- 명확한 운영 측면에서의 판단을 위해서는 각종 품질 조건을 같이 넣어서 판단을 해야 한다.
- 문제가 발생하여 복원할 때에는 manifest정보를 두어 복원 기준을 두도록 한다.

## 코드/예시
```python
# langfuse 피드백 스코어 예제
"""3강: 실제 Trace에 수업용 모의 피드백을 연결해요. 실제 사용자가 남긴 평가가 아니에요."""

from course import connect, read, save, run


def main():
    lf, _ = connect()
    # 앞 강의에서 이미 실행한 Trace를 사용해요. 이 파일은 모델을 다시 호출하지 않아요.
    rows = read("result_02.json")
    feedback = []
    # 네 요청 중 두 건에만 피드백이 있어요. 응답하지 않은 두 건을 만족으로 처리하지 않아요.
    for row, value, reason in [
        (rows[0], 0, "설명이 길다는 모의 의견"),
        (rows[1], 1, "간결하다는 모의 의견"),
    ]:
        # 점수는 질문 문장이 아니라 특정 실행의 trace_id에 연결해요.
        lf.create_score(
            trace_id=row["trace_id"],
            name="demo_feedback",
            value=value,
            data_type="BOOLEAN",
            comment="수업용 모의 피드백: " + reason,
        )
        feedback.append({"trace_id": row["trace_id"], "value": value, "reason": reason})
    lf.flush()
    save("result_03.json", {"simulation": True, "feedback": feedback})
    # 응답률 분모는 전체 요청, 긍정 비율 분모는 피드백을 준 요청이에요.
    print("피드백 응답률:", len(feedback) / len(rows))
    print("응답자 중 긍정 비율:", sum(x["value"] for x in feedback) / len(feedback))
    print("낮은 점수의 Trace에서 답변과 근거를 직접 확인하세요.")


if __name__ == "__main__":
    run(main)
```
```python
# langfuse를 통한 비용 산출 예제
"""4강: 서버의 비용을 재조회하고 변경 전후를 비교해요. 추가 모델 호출 없음."""

from datetime import datetime, timedelta, timezone
from time import sleep
from course import connect, read, save, run
from metrics import summarize


def main():
    lf, _ = connect()
    rows = read("result_02.json")
    # 로컬의 Trace ID로 서버 관측을 읽어 비용을 채워요. 다른 시간대나 버전의 실행을 섞지 않아요.
    for row in rows:
        observations = lf.api.observations.get_many(
            trace_id=row["trace_id"],
            limit=30,
            fields="core,basic,usage,model",
            from_start_time=datetime.fromisoformat(row["started_at"]) - timedelta(minutes=1),
            to_start_time=datetime.now(timezone.utc) + timedelta(minutes=1),
        ).data
        # 부모 span과 자식 모델 비용을 함께 더하면 중복될 수 있어 모델 구간만 골라요.
        generations = [o for o in observations if o.type.upper() == "GENERATION"]
        # 비용이 0이면 알려진 0이에요. None이면 아직 집계되지 않았거나 가격을 모르는 상태예요.
        costs = [o.total_cost for o in generations]
        row["cost_usd"] = sum(costs) if costs and all(c is not None for c in costs) else None
        sleep(2.1)  # 무료 계정의 조회 API 제한을 넘지 않도록 요청 간격을 둬요.
    # 같은 app_version끼리 집계해 before와 after를 분리해요.
    summary = {
        v: summarize([r for r in rows if r["app_version"] == v]) for v in ("before", "after")
    }
    for version, values in summary.items():
        print(version, values)
    # 표본 2개로 SLA나 통계적 성능 향상을 주장하지 않아요. 답의 내용과 출처도 함께 확인해요.
    save(
        "result_04.json",
        {"rows": rows, "summary": summary, "decision": "REVIEW", "reason": "버전당 2건"},
    )
    print("판정: REVIEW / 입력 토큰 감소와 답변 품질을 함께 검토하세요.")


if __name__ == "__main__":
    run(main)
```
